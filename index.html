<!-- 1. 引入样式和脚本 (零依赖) -->
<link rel="stylesheet" href="google-login.css">
<script src="google-login.js"></script>

<!-- 2. 页面中放置挂载容器 -->
<div id="google-login-container"></div>

<!-- 3. 初始化组件 -->
<script>
  GoogleLogin.embed('#google-login-container', {
    blogName: '我的极客博客',
    lang: 'zh-CN',    // 'zh-CN' | 'en' | 'ja'
    theme: 'auto',     // 'light' | 'dark' | 'auto'
    enable2FA: true,   // 是否开启 Google 手机两步验证流程
    onSuccess: function(response) {
      console.log('登录成功:', response.user);
      localStorage.setItem('blog_user', JSON.stringify(response.user));
    }
  });
</script>

<!-- 适用于 Hexo / Hugo / Typecho / WordPress 评论区 -->
<button onclick="triggerGoogleAuth()">Google 快捷登录评论</button>

<script>
function triggerGoogleAuth() {
  GoogleLogin.open({
    blogName: document.title,
    onSuccess: function(res) {
      // 自动回填博客评论区的昵称和邮箱输入框
      const emailField = document.querySelector('input[name="mail"], #mail, #email');
      const authorField = document.querySelector('input[name="author"], #author, #nick');
      if (emailField) emailField.value = res.user.email;
      if (authorField) authorField.value = res.user.name;
    }
  });
}
</script>
